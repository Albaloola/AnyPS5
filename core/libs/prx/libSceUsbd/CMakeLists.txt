function(add_sce_usbd_backend)
    set(featureChecks
        HAVE_CLOCK_GETTIME HAVE_PTHREAD_CONDATTR_SETCLOCK HAVE_PTHREAD_SETNAME_NP
        HAVE_PTHREAD_THREADID_NP HAVE_EVENTFD HAVE_PIPE2 HAVE_SYSLOG HAVE_ASM_TYPES_H
        HAVE_STRING_H HAVE_SYS_TIME_H HAVE_TIMERFD HAVE_NFDS_T HAVE_STRUCT_TIMESPEC HAVE_VISIBILITY)
    foreach(feature IN LISTS featureChecks)
        if(DEFINED CACHE{${feature}})
            set(had_${feature} TRUE)
            get_property(prior_${feature}_value CACHE ${feature} PROPERTY VALUE)
            get_property(prior_${feature}_type CACHE ${feature} PROPERTY TYPE)
            get_property(prior_${feature}_help CACHE ${feature} PROPERTY HELPSTRING)
        endif()
        unset(${feature} CACHE)
        unset(${feature})
    endforeach()
    add_subdirectory(${CMAKE_SOURCE_DIR}/3rdparty/libusb ${CMAKE_BINARY_DIR}/3rdparty/libusb EXCLUDE_FROM_ALL)
    foreach(feature IN LISTS featureChecks)
        unset(${feature} CACHE)
        if(had_${feature})
            set(${feature} "${prior_${feature}_value}" CACHE ${prior_${feature}_type} "${prior_${feature}_help}" FORCE)
        endif()
    endforeach()
endfunction()
add_sce_usbd_backend()
